#!/bin/bash
set -e

# === CONFIGURATION ===
DISTRO="bullseye"
ARCH="amd64"
LIVE_DIR="live-env"
MIRROR="http://deb.debian.org/debian"
ISO_NAME="ghostcore-rescue.iso"

echo "[*] Creating bootstrap directory: $LIVE_DIR"
mkdir -p $LIVE_DIR

# === BOOTSTRAP DEBIAN ===
echo "[*] Bootstrapping Debian base system..."
debootstrap --arch=$ARCH $DISTRO $LIVE_DIR $MIRROR

# === CHROOT SETUP ===
echo "[*] Mounting system directories..."
mount --bind /dev $LIVE_DIR/dev
mount --bind /proc $LIVE_DIR/proc
mount --bind /sys $LIVE_DIR/sys

# === INSTALL PACKAGES ===
cat > $LIVE_DIR/bootstrap_install.sh << 'EOF'
#!/bin/bash
set -e

echo "[*] Updating package list..."
apt update

echo "[*] Installing tools..."
apt install -y \
    python3 python3-pip python3-tk \
    wget curl git nano less openssh-client \
    ntfs-3g lsof htop yara \
    build-essential libyara-dev unzip

echo "[*] Installing Python packages..."
pip3 install psutil yara-python rich volatility3

echo "[*] Creating forensic tools directory..."
mkdir -p /opt/ghostcore/tools
EOF

chmod +x $LIVE_DIR/bootstrap_install.sh

# === ENTER CHROOT AND INSTALL ===
echo "[*] Entering chroot to install packages..."
chroot $LIVE_DIR /bootstrap_install.sh

# === CLEANUP CHROOT ===
echo "[*] Cleaning up..."
umount $LIVE_DIR/dev
umount $LIVE_DIR/proc
umount $LIVE_DIR/sys
rm $LIVE_DIR/bootstrap_install.sh

# === CREATE ISO ===
echo "[*] Creating bootable ISO..."
grub-mkrescue -o $ISO_NAME $LIVE_DIR

echo "[✓] ISO created: $ISO_NAME"

#!/usr/bin/env python3
import psutil
import socket
import subprocess
import time
import re
from datetime import datetime
from pathlib import Path

SUSPICIOUS_PORTS = {137, 138, 139, 445}
LOG_FILE = "/var/log/ghostcore/netbios_watch.log"
BLOCK_NETBIOS = True  # Set to False if you want to just monitor
ALERT_ON_SMB_TRANSFER = True
SMB_MOUNT_PATH = "/mnt/smbscan"

def log_event(message):
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    entry = f"[{timestamp}] {message}"
    print(entry)
    with open(LOG_FILE, "a") as f:
        f.write(entry + "\n")

def block_ports():
    for port in SUSPICIOUS_PORTS:
        subprocess.call([
            "iptables", "-A", "INPUT", "-p", "tcp", "--dport", str(port), "-j", "DROP"
        ])
        subprocess.call([
            "iptables", "-A", "INPUT", "-p", "udp", "--dport", str(port), "-j", "DROP"
        ])
    log_event("NetBIOS ports blocked via iptables")

def monitor_connections():
    conns = psutil.net_connections()
    for conn in conns:
        if conn.status != 'LISTEN' and conn.laddr and conn.raddr:
            if conn.laddr.port in SUSPICIOUS_PORTS or conn.raddr.port in SUSPICIOUS_PORTS:
                log_event(f"Suspicious connection detected: {conn.laddr} <-> {conn.raddr} [PID: {conn.pid}]")

def scan_for_open_shares():
    result = subprocess.run(["nmblookup", "-A", "127.0.0.1"], capture_output=True, text=True)
    if "MAC Address" in result.stdout and "GROUP" in result.stdout:
        log_event("NetBIOS names/groups detected locally — potential file sharing active.")

def detect_mounted_smb():
    output = subprocess.run(["mount"], capture_output=True, text=True).stdout

✅ PART 1: Windows CLSID Dump Utility (Offline Registry Parser)

🔧 Script: clsid_dump.py
#!/usr/bin/env python3
import os
import sys
import re
from pathlib import Path
import subprocess

REG_FILE = "/mnt/windows/Windows/System32/config/SOFTWARE"
DUMP_OUTPUT = "clsid_dump.txt"

def parse_registry_clsid():
    print("[*] Dumping Windows CLSIDs from offline registry...")

    if not Path(REG_FILE).exists():
        print("[!] Registry file not found. Is Windows mounted at /mnt/windows?")
        sys.exit(1)

    output = subprocess.run([
        "reged", "-x", REG_FILE, 
        "Microsoft\\Windows\\CurrentVersion\\Explorer\\ControlPanel\\NameSpace"
    ], capture_output=True, text=True)

    if output.returncode != 0:
        print("[!] Error reading registry:", output.stderr)
        sys.exit(1)

    raw = output.stdout
    matches = re.findall(r'\{[0-9A-Fa-f\-]{36}\}', raw)
    clsids = sorted(set(matches))

    with open(DUMP_OUTPUT, "w") as f:
        for clsid in clsids:
            f.write(clsid + "\n")

    print(f"[✓] Dumped {len(clsids)} CLSIDs to {DUMP_OUTPUT}")

if __name__ == "__main__":
    parse_registry_clsid()

⚠️ Short answer: Not safely — unless that file was already a valid ISO image.

🧠 Why Simply Renaming a File to .iso is Not Safe
🚫 Renaming file.ext → file.iso:
Does not convert it into a real ISO 9660 or UDF disk image

The file system inside won’t match the boot structure

Rufus or your BIOS/UEFI will fail to boot from it, or treat it as a corrupted image

✅ What is an ISO?
An .iso file is a sector-by-sector image of a CD/DVD or bootable volume. It contains:

A valid file system (ISO9660, Joliet, UDF, or hybrid)

Boot information (eltorito boot record or EFI boot image)

Organized binary structure, not just files with an .iso name

✅ When You Can Rename Safely
You can safely rename to .iso only if:

The file was created as an ISO using tools like mkisofs, grub-mkrescue, xorriso, etc.

Example: If you ran:

grub-mkrescue -o ghostcore.iso ghostcore-env/
You could rename ghostcore.iso to ghostcore-rescue.iso — and it would still be valid.

✅ How to Properly Create a Bootable ISO
From your live environment or builder script:

grub-mkrescue -o ghostcore.iso live-env/
Or use:

xorriso -as mkisofs -o ghostcore.iso -isohybrid-mbr /usr/lib/ISOLINUX/isohdpfx.bin \
  -c boot.cat -b isolinux/isolinux.bin -no-emul-boot -boot-load-size 4 \
  -boot-info-table live-env/
Or on Windows:

Use ImgBurn, [PowerISO], or [UltraISO] to make a proper bootable ISO from files

Don't just rename .zip or .exe to .iso

🔚 TL;DR
Rename to .iso?	Safe?	Why
.zip → .iso	❌	Invalid structure
.exe → .iso	❌	Not a disk image
.img → .iso	⚠️	Depends — .img might be raw block image, but not bootable ISO
Real .iso → renamed	✅	Safe if built correctly

#!/usr/bin/env python3
import os
import subprocess
import tkinter as tk
from tkinter import ttk, messagebox

ISO_FILE = "ghostcore-rescue.iso"  # Make sure this exists in the same directory

def list_usb_devices():
    try:
        result = subprocess.check_output(
            "lsblk -dpno NAME,SIZE,MODEL,TRAN | grep -E 'usb|removable|sd[a-z]$'",
            shell=True,
            universal_newlines=True
        )
        devices = result.strip().split('\n')
        return [line.strip() for line in devices if line.strip()]
    except subprocess.CalledProcessError:
        return []

def flash_iso(device):
    try:
        cmd = ["dd", f"if={ISO_FILE}", f"of={device}", "bs=4M", "status=progress", "oflag=sync", "conv=fsync"]
        subprocess.run(cmd, check=True)
        messagebox.showinfo("Success", f"ISO successfully written to {device}")
    except subprocess.CalledProcessError as e:
        messagebox.showerror("Error", f"Failed to write ISO: {e}")

def on_flash():
    selected = combo.get()
    if not selected:
        messagebox.showwarning("No Device", "Please select a USB device.")
        return
    device = selected.split()[0]
    confirm = messagebox.askyesno("Confirm Write", f"WARNING:\nAll data on {device} will be destroyed.\n\nProceed?")
    if confirm:
        flash_iso(device)

# GUI Setup
root = tk.Tk()
root.title("GhostCore ISO Flasher")
root.geometry("550x200")

tk.Label(root, text="Select USB Drive to Flash GhostCore ISO:", font=("Arial", 12)).pack(pady=10)

devices = list_usb_devices()
if not devices:
    messagebox.showerror("No USB Devices", "No removable USB devices detected.")
    root.destroy()
    exit()

combo = ttk.Combobox(root, values=devices, width=80)
combo.pack(pady=5)

tk.Button(root, text="Flash ISO", command=on_flash, bg="firebrick", fg="white", font=("Arial", 11)).pack(pady=15)

root.mainloop()

def list_usb_drives():
    drives = win32api.GetLogicalDriveStrings()
    drives = drives.split('\000')[:-1]
    usb_drives = []

    for drive in drives:
        try:
            drive_type = win32api.GetDriveType(drive)
            if drive_type == 2:  # DRIVE_REMOVABLE
