def parse_auth_results(val: str) -> dict:
    # Minimal parsing for SPF/DKIM/DMARC hints from Authentication-Results
    out = {}
    if not val:
        return out
    parts = [p.strip() for p in val.split(";")]
    for p in parts:
        if p.lower().startswith("spf="):
            out["spf"] = p.split("=", 1)[1]
        if p.lower().startswith("dkim="):
            out["dkim"] = p.split("=", 1)[1]
        if p.lower().startswith("dmarc="):
            out["dmarc"] = p.split("=", 1)[1]
    return out

@app.post("/analyze")
async def analyze(file: UploadFile = File(...)):
    if not file.filename.lower().endswith((".eml", ".txt", ".msg")):
        raise HTTPException(status_code=400, detail="Upload a raw RFC822 .eml/.msg")
    content = await file.read()
    try:
        msg = BytesParser(policy=policy.default).parsebytes(content)
    except Exception as e:
        raise HTTPException(status_code=400, detail=f"Parse error: {e}")

    headers = {k: v for (k, v) in msg.items()}
    received = msg.get_all("Received", [])
    authres_all = msg.get_all("Authentication-Results", []) or []
    auth = {}
    for ar in authres_all:
        auth.update(parse_auth_results(ar))

    return {
        "subject": msg.get("Subject"),
        "from": msg.get("From"),
        "to": msg.get("To"),
        "date": msg.get("Date"),
        "received_hops": received,
        "authentication_results": auth,
        "suspicion_signals": {
            "missing_spf": "spf" not in auth,
            "missing_dkim": "dkim" not in auth,
            "missing_dmarc": "dmarc" not in auth,
        }
    }

def parse_auth_results(val: str) -> dict:
    out = {}
    if not val:
        return out
    parts = [p.strip() for p in val.split(";")]
    for p in parts:
        if p.lower().startswith("spf="):
            out["spf"] = p.split("=", 1)[1]
        if p.lower().startswith("dkim="):
            out["dkim"] = p.split("=", 1)[1]
        if p.lower().startswith("dmarc="):
            out["dmarc"] = p.split("=", 1)[1]

A_TAG_PATTERN = re.compile(r'(<a\s[^>]*href=[\'"])([^\'"]+)([\'"][^>]*>)', re.IGNORECASE)

def rewrite_links(html: str, campaign_id: str, recipient: str) -> str:
    def repl(match):
        prefix, original_url, suffix = match.groups()
        safe_url = f"{LINK_GATEWAY}/c/{campaign_id}/{recipient}"
        return f'{prefix}{safe_url}?orig={requests.utils.quote(original_url)}{suffix}'
    return A_TAG_PATTERN.sub(repl, html)

def build_email(req: DispatchRequest, recipient: str) -> EmailMessage:
    msg = EmailMessage()
    subj = req.subject if req.subject.startswith("[TRAINING]") else f"[TRAINING] {req.subject}"
    msg["Subject"] = subj
    msg["To"] = recipient
    msg["From"] = "training@eidolon.local"
    safe_html = WATERMARK_BANNER + rewrite_links(req.html_body, req.campaign_id, recipient)
    msg.set_content("This is a training simulation. View in HTML.")
    msg.add_alternative(safe_html, subtype="html")
    msg["X-Eidolon-Template"] = req.template_name
    msg["X-Eidolon-Campaign"] = req.campaign_id
    msg["X-Eidolon-Classification"] = "training"
    return msg

        self.dry_run = dry_run

    def crawl_root(self, root_url: str):
        # synchronous wrapper for asyncio run
        return asyncio.run(self._crawl(root_url))

    async def _crawl(self, root_url: str):
        queue = asyncio.Queue()
        await queue.put((root_url, 0))
        visited = set()
        results = []

        timeout = aiohttp.ClientTimeout(total=30)
        connector = aiohttp.TCPConnector(limit_per_host=self.concurrency)
        proxy = self.burp.proxy if self.burp else None

        headers = {"User-Agent": self.user_agent}

        async with aiohttp.ClientSession(timeout=timeout, connector=connector) as session:
            sem = asyncio.Semaphore(self.concurrency)
            tasks = []
            while not queue.empty() and len(visited) < self.request_budget:
                url, depth = await queue.get()
                if url in visited or depth > self.max_depth:
                    continue
                if not self._in_scope(url):
                    continue
                visited.add(url)
                tasks.append(asyncio.create_task(self._fetch(session, url, depth, headers, proxy, results, queue)))
            await asyncio.gather(*tasks)
        return results

    async def _fetch(self, session, url, depth, headers, proxy, results, queue):
        try:
            if self.dry_run:
                results.append(Page(url=url, status=0, depth=depth, content_type="dry-run"))
                return
            async with session.get(url, headers=headers, proxy=proxy) as resp:
                ct = resp.headers.get("Content-Type", "")
                html = await resp.text(errors="ignore") if "text/html" in ct else None
                results.append(Page(url=url, status=resp.status, depth=depth, content_type=ct, html=html))
                if html and depth < self.max_depth:
                    for link in self._extract_links(url, html):
                        await queue.put((link, depth + 1))
        except Exception:
            # keep minimal logging here; expand in production
            return

    def _extract_links(self, base_url: str, html: str):
        # Minimal link extraction; replace with robust parser as needed
        import re
        links = set()
        for m in re.finditer(r'href=["\']([^"\']+)["\']', html, re.IGNORECASE):
            link = urljoin(base_url, m.group(1))
            if self._in_scope(link):
                links.add(link)
        return links

    def _in_scope(self, url: str) -> bool:
        if any(p.search(url) for p in self.exclude_patterns):
            return False
        return any(p.search(url) for p in self.include_patterns)

    def _extract_endpoints(self, page: Page) -> List[Endpoint]:
        # rudimentary endpoint extraction from JS fetch/xhr patterns
        import re
        endpoints = []
        for m in re.finditer(r'(fetch|xhr|axios)\(["\']([^"\']+)["\']', page.html, re.IGNORECASE):
            endpoints.append(Endpoint(method="GET", url=m.group(2), params={}))
        return endpoints

    def _extract_forms(self, page: Page) -> List[Form]:
        import re
        forms = []
        for fm in re.finditer(r"<form[^>]*>(.*?)</form>", page.html, re.IGNORECASE | re.DOTALL):
            method_m = re.search(r'method=["\'](\w+)["\']', fm.group(0), re.IGNORECASE)
            action_m = re.search(r'action=["\']([^"\']+)["\']', fm.group(0), re.IGNORECASE)
            inputs = []
            for im in re.finditer(r'<input[^>]*name=["\']([^"\']+)["\'][^>]*>', fm.group(1), re.IGNORECASE):
                inputs.append({"name": im.group(1)})
            forms.append(Form(action=action_m.group(1) if action_m else page.url,
                              method=(method_m.group(1) if method_m else "GET").upper(),
                              inputs=inputs))
        return forms

    def _extract_scripts(self, page: Page) -> List[Script]:
        import re
        scripts = []
        for sm in re.finditer(r'<script[^>]*src=["\']([^"\']+)["\']', page.html, re.IGNORECASE):
            scripts.append(Script(url=sm.group(1), inline=False))
        for sm in re.finditer(r'<script[^>]*>(.*?)</script>', page.html, re.IGNORECASE | re.DOTALL):
            scripts.append(Script(url=page.url + "#inline", inline=True))
        return scripts

⚠️ Security & Abuse Awareness
(Important — Read Carefully)
❗ SMS‑to‑Email Gateways Are Commonly Abused
Carrier email gateways have historically been abused for:

Spam delivery

Phishing attempts

Social engineering

Bypassing traditional SMS APIs

Alert fatigue attacks

Identity confusion (messages appear as “SMS” but originate from email)

This project exists in part to help teams understand and defend against those risks.

🚨 Common Abuse Patterns (High‑Level, Defensive)
⚠️ This section is educational, not instructional.

Spoofed alerts
Messages crafted to look like bank, carrier, or security alerts.

Out‑of‑band phishing
SMS used to bypass email filters and push victims toward malicious links.

Carrier gateway blind spots
Gateways often lack strong sender authentication or rate limiting.

Message fragmentation abuse
Long messages split into multiple SMS parts to evade filters.

MMS misuse
MMS used to carry links or payload references when SMS filters are stricter.

🛡️ Defensive Guidance
If you operate or monitor SMS/email systems:

Log and monitor outbound email to known carrier gateways

Rate‑limit messages per destination

Enforce sender allowlists

Add human‑readable banners or prefixes to alerts

Train users that SMS ≠ trusted

Use this tool in controlled lab or training environments only

🔐 Responsible Use Policy
This project must not be used to:

Send unsolicited messages

Conduct harassment or spam campaigns

Bypass carrier protections

Impersonate real organizations

Evade consent or opt‑out mechanisms

Use is restricted to:

Systems you own

Users who have opted in
